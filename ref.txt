        function toggleTab(tab) {
            const indicator = document.getElementById('tab-indicator');
            const globalBtn = document.getElementById('tab-global');
            const friendsBtn = document.getElementById('tab-friends');

            if (tab === 'global') {
                indicator.style.transform = 'translateX(0)';
                globalBtn.classList.remove('text-on-surface-variant');
                globalBtn.classList.add('text-on-surface');
                friendsBtn.classList.remove('text-on-surface');
                friendsBtn.classList.add('text-on-surface-variant');
            } else {
                indicator.style.transform = 'translateX(100%)';
                friendsBtn.classList.remove('text-on-surface-variant');
                friendsBtn.classList.add('text-on-surface');
                globalBtn.classList.remove('text-on-surface');
                globalBtn.classList.add('text-on-surface-variant');
            }
        }
